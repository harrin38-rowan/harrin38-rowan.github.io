#Net migration

Net Migration is given by the following formula: \[ \text{Migration into country} - \text{Migration out of country} \]

#|code-fold: true
#|llayout-ncol: 2

least = df |>
  arrange(Net.migration) |>
  head(3) |>
  select("Most Negative Net Migration" = Country)

reactable(least)
